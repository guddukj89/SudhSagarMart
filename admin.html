/* ===== PRODUCT ADD FUNCTION ===== */ const productInputs = document.querySelectorAll('#products input'); const productCategory = document.querySelector('#products select'); const addProductBtn = document.querySelector('#products button'); if (addProductBtn) { addProductBtn.addEventListener('click', function () { const name = productInputs[0]?.value.trim(); const price = productInputs[1]?.value.trim(); const oldPrice = productInputs[2]?.value.trim(); const stock = productInputs[3]?.value.trim(); const category = productCategory?.value || 'General'; const seller = productInputs[4]?.value.trim(); if (!name || !price || !stock || !seller) { alert('Product Name, Price, Stock aur Seller Name bhariye.'); return; } const product = { id: Date.now(), name: name, price: price, oldPrice: oldPrice, stock: stock, category: category, seller: seller, status: 'Active' }; let products = JSON.parse(localStorage.getItem('sudhsagar_products') || '[]'); products.push(product); localStorage.setItem( 'sudhsagar_products', JSON.stringify(products) ); addProductToTable(product); productInputs.forEach(input => input.value = ''); if (productCategory) { productCategory.selectedIndex = 0; } alert('✅ Product successfully added!'); }); } function addProductToTable(product) { const table = document.querySelector('#products table'); if (!table) return; let tbody = table.querySelector('tbody'); if (!tbody) { tbody = document.createElement('tbody'); table.appendChild(tbody); } const row = document.createElement('tr'); row.innerHTML = ` ${product.name} ${product.seller} ₹${product.price} ${product.stock} Active `; tbody.appendChild(row); }
